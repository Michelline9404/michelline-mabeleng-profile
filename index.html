/* ============================================================ INTERACTIVE FEATURES FOR THE CV PAGE ============================================================ */ /** * 1. THEME TOGGLE * Switches between light and dark mode by toggling * the `data-theme` attribute on and saving the * preference to localStorage. */ function toggleTheme() { const root = document.documentElement; const current = root.getAttribute('data-theme'); const next = current === 'dark' ? 'light' : 'dark'; root.setAttribute('data-theme', next); localStorage.setItem('cv-theme', next); // Update the button label const btn = document.querySelector('.btn-toggle-theme'); if (btn) { btn.textContent = next === 'dark' ? '☀️ Light Mode' : '🌙 Dark Mode'; } } /** * Apply saved theme on page load. * Falls back to the user's OS preference if nothing is saved. */ (function applySavedTheme() { const saved = localStorage.getItem('cv-theme'); const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches; const theme = saved || (prefersDark ? 'dark' : 'light'); document.documentElement.setAttribute('data-theme', theme); // Set button label after DOM is ready window.addEventListener('DOMContentLoaded', () => { const btn = document.querySelector('.btn-toggle-theme'); if (btn) { btn.textContent = theme === 'dark' ? '☀️ Light Mode' : '🌙 Dark Mode'; } }); })(); /** * 2. DOWNLOAD PDF * Uses the browser's print-to-PDF functionality. * For a true PDF, this triggers the print dialog where the * user can choose "Save as PDF". */ document.addEventListener('DOMContentLoaded', () => { const downloadBtn = document.getElementById('downloadBtn'); if (downloadBtn) { downloadBtn.addEventListener('click', () => { // Small delay ensures the button click animation completes setTimeout(() => window.print(), 150); }); } }); /** * 3. SCROLL-TRIGGERED ANIMATIONS * Adds a subtle fade-up effect to sections as they enter * the viewport. Uses IntersectionObserver for performance. */ document.addEventListener('DOMContentLoaded', () => { const sections = document.querySelectorAll('.section'); if (!('IntersectionObserver' in window)) { // Fallback: just show everything sections.forEach(s => (s.style.opacity = '1')); return; } const observer = new IntersectionObserver( (entries) => { entries.forEach((entry) => { if (entry.isIntersecting) { entry.target.style.opacity = '1'; entry.target.style.transform = 'translateY(0)'; observer.unobserve(entry.target); } }); }, { threshold: 0.08, rootMargin: '0px 0px -40px 0px' } ); sections.forEach((section) => { // Start hidden – CSS animation will handle the reveal section.style.opacity = '0'; section.style.transform = 'translateY(18px)'; section.style.transition = 'opacity 0.5s ease, transform 0.5s ease'; observer.observe(section); }); }); /** * 4. PRINT SHORTCUT * Allows pressing Ctrl+P / Cmd+P naturally (no override needed), * but we ensure the print styles are applied correctly. */ window.addEventListener('beforeprint', () => { // Ensure all sections are visible before printing document.querySelectorAll('.section').forEach((s) => { s.style.opacity = '1'; s.style.transform = 'none'; }); });
